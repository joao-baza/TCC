\begin{frame}{Base computacional reutilizavel}
  \begin{itemize}
    \item Um nucleo de rotinas atende varios modulos sem reescrever formulas
    \item Separacao entre calculo, interface e apoio didatico reduz acoplamento
    \item Componentes compartilhados permitem reaproveitar estruturas entre temas
    \item Novos modulos podem crescer por extensao, sem duplicar a base existente
  \end{itemize}
\end{frame}

\begin{frame}{API, validacao e apoio didatico}
  \begin{itemize}
    \item Contratos de entrada e saida padronizam como a interface consome os calculos
    \item A validacao centralizada bloqueia combinacoes inconsistentes antes do processamento
    \item Exemplos prontos ajudam a iniciar cenarios coerentes na interface
    \item Glossario e explicacoes apoiam a leitura tecnica dos resultados
  \end{itemize}
\end{frame}

\begin{frame}{Deploy e distribuicao}
  \begin{itemize}
    \item Aplicacao web publicada para uso imediato
    \item Execucao local permite reproduzir resultados e inspecionar o codigo aberto
    \item Empacotamento desktop demonstra portabilidade da mesma solucao
    \item CI/CD e testes sustentam a manutencao continua do projeto
  \end{itemize}
\end{frame}
