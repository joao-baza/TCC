\begin{frame}{A arquitetura precisa preservar a cadeia técnica}
  \begin{columns}[T,onlytextwidth]
    \column{0.32\textwidth}
      \begin{block}{Hipótese}
        O domínio de validade vem antes do cálculo: fase, regime, unidade, catálogo e condição operacional.
      \end{block}
    \column{0.32\textwidth}
      \begin{block}{Modelo}
        O backend aplica equações, propriedades e métodos numéricos sem duplicar matemática no frontend.
      \end{block}
    \column{0.32\textwidth}
      \begin{block}{Interpretação}
        A mesma resposta alimenta tabela, gráfico, validação e explicação didática.
      \end{block}
  \end{columns}
  \vspace{0.55em}
  \begin{alertblock}{Critério de projeto}
    Equação, unidade, gráfico e explicação não podem divergir entre camadas.
  \end{alertblock}
\end{frame}

\begin{frame}{A arquitetura separa interface, modelo e distribuição}
  \begin{columns}[T,onlytextwidth]
    \column{0.32\textwidth}
      \begin{block}{Frontend}
        Organiza formulário, resultado, gráfico, glossário e exemplos sem recalcular a matemática central.
      \end{block}
    \column{0.32\textwidth}
      \begin{block}{Backend}
        FastAPI, Pydantic, Pint, CoolProp, NumPy e SciPy concentram modelos, unidades e validações.
      \end{block}
    \column{0.32\textwidth}
      \begin{block}{Entrega}
        Web publicada, execução local, Docker, CI/CD e desktop reaproveitam a mesma base validada.
      \end{block}
  \end{columns}
\end{frame}

\begin{frame}{A API também monta visualizações}
  \begin{columns}[T,onlytextwidth]
    \column{0.54\textwidth}
      \begin{itemize}
        \item Endpoints retornam eixos, séries, marcadores e metadados junto ao resultado calculado.
        \item O frontend renderiza o contrato sem reconstruir curvas com lógica paralela.
        \item Isso mantém coerência entre valor numérico, curva e explicação didática.
      \end{itemize}
    \column{0.40\textwidth}
      \begin{block}{Exemplos validados no código}
        Moody, curva de bomba, NPSH, altura manométrica, T-x-y, McCabe-Thiele, envelope de fase, Arrhenius, PFR e balanço.
      \end{block}
  \end{columns}
\end{frame}

\begin{frame}{Fundamentos hidráulicos implementados}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \begin{block}{Continuidade}
        \[
          D = \sqrt{\frac{4Q}{\pi V}}
        \]
      \end{block}
      \begin{block}{Reynolds}
        \[
          Re = \frac{\rho V D}{\mu}
        \]
      \end{block}
    \column{0.48\textwidth}
      \begin{block}{Darcy-Weisbach}
        \[
          h_f = f\frac{L+\sum L_{eq}}{D}\frac{V^2}{2g}
        \]
      \end{block}
      \begin{block}{Leitura didática}
        Regime, rugosidade, perda de carga e seleção de bomba são apresentados juntos.
      \end{block}
  \end{columns}
\end{frame}

\begin{frame}{Reatores unem cinética e estequiometria}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \begin{block}{Lei de velocidade}
        \[
          r = k \prod C_i^{n_i}
        \]
      \end{block}
      \begin{block}{CSTR}
        \[
          V = \frac{F_{A0}X}{-r_A}
        \]
      \end{block}
    \column{0.48\textwidth}
      \begin{block}{PFR}
        \[
          V = F_{A0}\int_0^X \frac{dX}{-r_A}
        \]
      \end{block}
      \begin{block}{Visualizações e hipótese}
        Levenspiel, Arrhenius e perfis espaciais ajudam a interpretar volume e conversão; no escopo atual, os reatores são isotérmicos e monofásicos.
      \end{block}
  \end{columns}
\end{frame}

\begin{frame}{Termodinâmica e separações ficam exploráveis}
  \begin{itemize}
    \item CoolProp fornece propriedades de fluidos puros, misturas e limites de fase.
    \item Diagramas T-x-y e McCabe-Thiele tornam visível a construção de estágios teóricos.
    \item Envelope de fase e pressão de vapor apoiam discussões de mudança de fase e cavitação.
    \item A superfície T-P mostra sensibilidade de propriedades a temperatura e pressão.
  \end{itemize}
\end{frame}
