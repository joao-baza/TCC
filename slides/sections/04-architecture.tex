\begin{frame}{Base computacional reutilizável}
  \begin{itemize}
    \item Um núcleo de rotinas atende vários módulos sem reescrever fórmulas
    \item Separação entre cálculo, interface e apoio didático reduz acoplamento
    \item Componentes compartilhados permitem reaproveitar estruturas entre temas
    \item Novos módulos podem crescer por extensão, sem duplicar a base existente
  \end{itemize}
\end{frame}

\begin{frame}{API, validação e apoio didático}
  \begin{itemize}
    \item Contratos de entrada e saída padronizam como a interface consome os cálculos
    \item A validação centralizada bloqueia combinações inconsistentes antes do processamento
    \item Exemplos prontos ajudam a iniciar cenários coerentes na interface
    \item Glossário e explicações apoiam a leitura técnica dos resultados
  \end{itemize}
\end{frame}

\begin{frame}{Deploy e distribuição}
  \begin{itemize}
    \item Aplicação web publicada para uso imediato
    \item Execução local permite reproduzir resultados e inspecionar o código aberto
    \item Empacotamento desktop demonstra portabilidade da mesma solução
    \item CI/CD e testes sustentam a manutenção contínua do projeto
  \end{itemize}
\end{frame}
