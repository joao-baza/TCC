\begin{frame}{Complexidade química: modelos diferentes}
  \small
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \begin{block}{Transporte e bombas}
        Regime de escoamento, rugosidade, diâmetro, fator de atrito, perda de carga, NPSH e altura manométrica dependem de relações acopladas.
      \end{block}
      \begin{block}{Propriedades}
        Temperatura, pressão, componente, mistura e mudança de fase alteram o caminho de cálculo e a interpretação.
      \end{block}
    \column{0.48\textwidth}
      \begin{block}{Reatores}
        Cinética, estequiometria, conversão, volume e perfis espaciais mudam conforme CSTR, PFR ou reciclo.
      \end{block}
      \begin{block}{Balanços}
        Correntes, reações, divisões e reciclos precisam preservar coerência material em todo o sistema.
      \end{block}
  \end{columns}
\end{frame}

\begin{frame}{Complexidade integrada: engenharia em software}
  \begin{columns}[T,onlytextwidth]
    \column{0.50\textwidth}
      \begin{itemize}
        \item O backend concentra modelos, unidades, validações e métodos numéricos.
        \item A API transforma cada cálculo em contrato reutilizável.
        \item A interface consome respostas estruturadas, não fórmulas duplicadas.
      \end{itemize}
    \column{0.44\textwidth}
      \begin{block}{Ponto crítico}
        A mesma resposta precisa servir ao cálculo, à validação, à tabela, ao gráfico e à explicação didática.
      \end{block}
      \begin{alertblock}{Resultado}
        O rigor químico precisa sobreviver à camada de software.
      \end{alertblock}
  \end{columns}
\end{frame}

\begin{frame}{Complexidade didática: rigor sem caixa-preta}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \begin{block}{Simplificar o uso}
        Exemplos coerentes, formulários guiados e mensagens de validação reduzem a barreira de entrada.
      \end{block}
      \begin{block}{Manter rastreabilidade}
        Unidades, hipóteses, variáveis e resultados continuam visíveis para discussão técnica.
      \end{block}
    \column{0.48\textwidth}
      \begin{block}{Explicar o fenômeno}
        Gráficos, glossário e blocos conceituais conectam o número à interpretação de engenharia.
      \end{block}
      \begin{alertblock}{Equilíbrio buscado}
        A interface ajuda o estudante sem esconder o raciocínio que sustenta o cálculo.
      \end{alertblock}
  \end{columns}
\end{frame}

\begin{frame}{A arquitetura precisa preservar a cadeia técnica}
  \begin{columns}[T,onlytextwidth]
    \column{0.32\textwidth}
      \begin{block}{Hipótese}
        O domínio de validade vem antes do cálculo: fase, regime, unidade, catálogo e condição operacional.
      \end{block}
    \column{0.32\textwidth}
      \begin{block}{Modelo}
        O backend aplica equações, propriedades e métodos numéricos sem duplicar matemática no frontend.
      \end{block}
    \column{0.32\textwidth}
      \begin{block}{Interpretação}
        A mesma resposta alimenta tabela, gráfico, validação e explicação didática.
      \end{block}
  \end{columns}
  \vspace{0.55em}
  \begin{alertblock}{Critério de projeto}
    Equação, unidade, gráfico e explicação não podem divergir entre camadas.
  \end{alertblock}
\end{frame}

\begin{frame}{A arquitetura separa interface, modelo e distribuição}
  \begin{columns}[T,onlytextwidth]
    \column{0.32\textwidth}
      \begin{block}{Frontend}
        Organiza formulário, resultado, gráfico, glossário e exemplos sem recalcular a matemática central.
      \end{block}
    \column{0.32\textwidth}
      \begin{block}{Backend}
        FastAPI, Pydantic, Pint, CoolProp, NumPy e SciPy concentram modelos, unidades e validações.
      \end{block}
    \column{0.32\textwidth}
      \begin{block}{Entrega}
        Web publicada, execução local, Docker, CI/CD e desktop reaproveitam a mesma base validada.
      \end{block}
  \end{columns}
\end{frame}

\begin{frame}{A API também monta visualizações}
  \begin{columns}[T,onlytextwidth]
    \column{0.54\textwidth}
      \begin{itemize}
        \item Endpoints retornam eixos, séries, marcadores e metadados junto ao resultado calculado.
        \item O frontend renderiza o contrato sem reconstruir curvas com lógica paralela.
        \item Isso mantém coerência entre valor numérico, curva e explicação didática.

        Moody, curva de bomba, NPSH, altura manométrica, T-x-y, McCabe-Thiele, envelope de fase, Arrhenius, PFR e balanço.
      \end{block}
  \end{columns}
\end{frame}

\begin{frame}{Fundamentos hidráulicos implementados}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \begin{block}{Continuidade}
        \[
          D = \sqrt{\frac{4Q}{\pi V}}
        \]
      \end{block}
      \begin{block}{Reynolds}
        \[
          Re = \frac{\rho V D}{\mu}
        \]
      \end{block}
    \column{0.48\textwidth}
      \begin{block}{Darcy-Weisbach}
        \[
          h_f = f\frac{L+\sum L_{eq}}{D}\frac{V^2}{2g}
        \]
      \end{block}
      \begin{block}{Leitura didática}

      \end{block}
  \end{columns}
\end{frame}

\begin{frame}{Reatores unem cinética e estequiometria}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \begin{block}{Lei de velocidade}
        \[
          r = k \prod C_i^{n_i}
        \]
      \end{block}
      \begin{block}{CSTR}
        \[
          V = \frac{F_{A0}X}{-r_A}
        \]
      \end{block}
    \column{0.48\textwidth}
      \begin{block}{PFR}
        \[
          V = F_{A0}\int_0^X \frac{dX}{-r_A}
        \]
      \end{block}
      \begin{block}{Visualizações e hipótese}
        Levenspiel, Arrhenius e perfis espaciais ajudam a interpretar volume e conversão; no escopo atual, os reatores são isotérmicos e monofásicos.
      \end{block}
  \end{columns}
\end{frame}
