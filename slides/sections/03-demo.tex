\begin{frame}{Roteiro da demonstração}
  \begin{enumerate}
    \item Posicionamento geral do DCOU
    \item Percurso sintético pelos módulos
    \item Recursos didáticos e visualizações
    \item Prova de execução online, local e Linux
  \end{enumerate}
\end{frame}

\begin{frame}{Checkpoints da demonstração}
  \begin{itemize}
    \item Home e proposta do sistema
    \item Módulos de engenharia em plataforma única
    \item Validação e apoio didático, não apenas número bruto
    \item Reprodutibilidade fora do ambiente publicado
  \end{itemize}
\end{frame}
