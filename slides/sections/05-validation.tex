\begin{frame}{Resultado técnico do TCC}
  \begin{itemize}
    \item O DCOU integra cálculo, validação e visualização em uma experiência única de estudo.
    \item A separação backend/frontend evita duplicação de fórmulas e centraliza regras de engenharia.
    \item A validação comparativa mostra concordância em casos representativos de hidráulica e reatores.
    \item A contribuição não é apenas funcional: é uma base aberta para explicar, auditar e estender cálculos de operações unitárias.
  \end{itemize}
\end{frame}

\begin{frame}{Limitações e continuidade}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \begin{block}{Limites atuais}
        \begin{itemize}
          \item Não substitui simuladores industriais completos.
          \item Ainda não cobre trocadores, absorção, secagem, multifásicos e múltiplas reações.
          \item Bancos de propriedades e catálogos podem crescer.
        \end{itemize}
      \end{block}
    \column{0.48\textwidth}
      \begin{block}{Próximos passos}
        \begin{itemize}
          \item Expandir módulos mantendo a arquitetura atual.
          \item Ampliar validações comparativas.
          \item Evoluir relatórios, rastreabilidade e integração CAPE-OPEN.
        \end{itemize}
      \end{block}
  \end{columns}
\end{frame}
