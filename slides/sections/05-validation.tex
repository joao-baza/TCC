\transitionframe{Evidências de confiabilidade}{Comparações, validações e integração entre as camadas}

\begin{frame}{Como a confiabilidade foi tratada}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \begin{block}{Validação de engenharia}
        Comparações com equações clássicas, literatura e resultados esperados para Reynolds, atrito, perda de carga, NPSH, diâmetros, reatores e balanços.
      \end{block}
    \column{0.48\textwidth}
      \begin{block}{Validação de software}
        Testes automatizados, validação Pydantic, contratos tipados, exemplos da API e execução local do frontend/backend.
      \end{block}
  \end{columns}
\end{frame}

\begin{frame}{Exemplos de concordância numérica}
  \begin{itemize}
    \item \textbf{Perda de carga:} casos de tubulação reta e com acessórios foram comparados com Darcy-Weisbach e Hazen-Williams.
    \item \textbf{Reynolds:} entradas por viscosidade dinâmica e cinemática retornaram valores coerentes.
    \item \textbf{Fator de atrito:} Colebrook-White, Swamee-Jain e Haaland foram conferidos.
    \item \textbf{NPSH e altura manométrica:} termos de Bernoulli foram comparados com cálculo manual.
    \item \textbf{Componentes:} água, etanol, metanol e outros componentes altamente estudados retornaram propriedades coerentes com referência termodinâmica.
  \end{itemize}
\end{frame}

\begin{frame}{Resultados do trabalho}
  \begin{itemize}
    \item A aplicação integra cálculo, validação e visualização em uma experiência única.
    \item A arquitetura separa regras de engenharia, apresentação e distribuição.
    \item A biblioteca de visualizações transforma resultados em objetos didáticos.
    \item A API aberta permite uso pela interface e também por outros consumidores.
    \item O projeto atinge o objetivo de entregar uma base gratuita, modular e extensível.
  \end{itemize}
\end{frame}

\begin{frame}{Limitações assumidas}
  \begin{itemize}
    \item O DCOU ainda não substitui simuladores industriais completos.
    \item Módulos como trocadores de calor, absorção, secagem, operações multifásicas e múltiplas reações ficam como evolução.
    \item Alguns dados de catálogo e propriedades ainda podem crescer para bancos externos.
  \end{itemize}
\end{frame}

\begin{frame}{Perspectivas de continuidade}
  \begin{itemize}
    \item Expandir módulos de operações unitárias mantendo a arquitetura atual.
    \item Ampliar bancos de propriedades, catálogos e validações comparativas.
    \item Evoluir exportação de relatórios e rastreabilidade dos cenários.
    \item Estudar integração futura com CAPE-OPEN e ferramentas externas.
    \item Reforçar colaboração acadêmica por meio do repositório aberto.
  \end{itemize}
\end{frame}
