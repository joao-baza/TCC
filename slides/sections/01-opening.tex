\begin{frame}{O problema não é só calcular}
  \vspace*{-0.85em}
  \begin{columns}[T,onlytextwidth]
    \column{0.52\textwidth}
      \begin{block}{Lacuna observada}
        {\small
        \begin{itemize}
          \setlength{\itemsep}{0.25em}
          \item Simuladores como Aspen Plus, HYSYS e ChemCAD são robustos, mas têm alto custo de acesso e licenciamento.
          \item Alternativas como DWSIM, Excel e MATLAB apoiam cálculos, mas nem sempre integram módulos e leitura didática.
          \item Exigem curva de aprendizado elevada ou apresentam interfaces defasadas.
          \item O estudante vê o resultado, porém nem sempre enxerga o fenômeno por trás dele.
        \end{itemize}
        }
      \end{block}
    \column{0.42\textwidth}
      \begin{alertblock}{Pergunta da defesa}
        {\small Como construir uma ferramenta aberta que una cálculo de engenharia, leitura didática e execução reprodutível?}
      \end{alertblock}
  \end{columns}
\end{frame}

\begin{frame}{A automação só ajuda quando preserva o raciocínio de engenharia}
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \begin{block}{Fenômeno e modelo}
        Operações unitárias exigem conectar balanços, propriedades, escoamento, cinética e equilíbrio a hipóteses de cálculo explícitas.
      \end{block}
      \begin{itemize}
        \item Unidade, variável e domínio de validade precisam continuar visíveis.
        \item O resultado numérico depende do fenômeno físico modelado.
      \end{itemize}
    \column{0.48\textwidth}
      \begin{alertblock}{Pergunta técnica}
        Como automatizar cálculos sem transformar o processo em uma caixa-preta?
      \end{alertblock}
      \begin{itemize}
        \item A interface deve apoiar a interpretação, não substituí-la.
        \item O código aberto permite auditoria e extensão acadêmica.
      \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{Objetivo e tese da defesa}
  \begin{block}{Objetivo da monografia}
    Desenvolver uma plataforma web aberta para cálculo e apoio didático em operações unitárias, integrando rotinas de engenharia, visualizações, validação de entrada e API pública.
  \end{block}
  \vspace{0.45em}
  \begin{columns}[T,onlytextwidth]
    \column{0.32\textwidth}
      \begin{block}{Cálculo}
        Tubulações, escoamento, bombas, propriedades, reatores e balanços.
      \end{block}
    \column{0.32\textwidth}
      \begin{block}{Interpretação}
        Hipóteses, unidades, exemplos, glossário e visualizações.
      \end{block}
    \column{0.32\textwidth}
      \begin{block}{Confiabilidade}
        Backend validado, contratos de API e execução reprodutível.
      \end{block}
  \end{columns}
