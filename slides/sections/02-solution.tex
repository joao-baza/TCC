\begin{frame}{O que é o DCOU}
  \begin{columns}[T,onlytextwidth]
    \column{0.46\textwidth}
      \begin{block}{Produto acadêmico}
        Plataforma web gratuita para cálculo e interpretação de operações unitárias em Engenharia Química.
      \end{block}
      \begin{itemize}
        \item Interface navegável por módulos.
        \item API pública documentada pelo FastAPI.
        \item Execução online e local; empacotamento desktop em estágio inicial.
      \end{itemize}
    \column{0.50\textwidth}
      \begin{block}{Modos de execução}
        \scriptsize
        \setlength{\tabcolsep}{4.5pt}
        \renewcommand{\arraystretch}{1.18}
        \begin{tabular*}{\linewidth}{@{\extracolsep{\fill}}>{\raggedright\arraybackslash}p{0.20\linewidth}>{\raggedright\arraybackslash}p{0.30\linewidth}>{\raggedright\arraybackslash}p{0.34\linewidth}}
          \textbf{Modo} & \textbf{Acesso} & \textbf{Uso} \\
          \hline
          Local & Backend local + navegador & Desenvolvimento e auditoria \\
          Online & Web publicada & Banca e demonstrações \\
          Desktop & Empacotado com backend local & Uso em máquina própria \\
        \end{tabular*}
      \end{block}
  \end{columns}
\end{frame}

\begin{frame}{Cobertura implementada}
  \footnotesize
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \begin{block}{Hidráulica e bombas}
        Tubulações, schedules, acessórios, diâmetro calculado/comercial, Reynolds, atrito, perda de carga, NPSH e altura manométrica.
      \end{block}
      \begin{block}{Propriedades e equilíbrios}
        Componentes, fluido puro, misturas, equilíbrio binário, envelope de fase, pressão de vapor e superfície T-P.
      \end{block}
    \column{0.48\textwidth}
      \begin{block}{Reatores}
        CSTR, PFR, reagente limitante, cinética de potência, Levenspiel, Arrhenius, perfis e reciclo.
      \end{block}
      \begin{block}{Balanço e apoio}
        Balanço com correntes, reações, splits, reciclo, gráficos, tabela, glossário e trilhas.
      \end{block}
      \begin{block}{Métodos e visualizações}
        Diagramas T-x-y, McCabe-Thiele, Moody, Levenspiel e curvas de sistema apoiam a interpretação dos cálculos.
      \end{block}
  \end{columns}
\end{frame}

\begin{frame}{O diferencial didático é a integração}
  \begin{columns}[T,onlytextwidth]
    \column{0.30\textwidth}
      \begin{block}{Entrada}
        Formulários com unidades, catálogos e exemplos coerentes.
      \end{block}
    \column{0.30\textwidth}
      \begin{block}{Processamento}
        Backend resolve cálculos, valida hipóteses e monta modelos de gráfico.
      \end{block}
    \column{0.30\textwidth}
      \begin{block}{Interpretação}
        Tabelas, gráficos, glossário e blocos ``Como funciona'' conectam número e conceito.
      \end{block}
  \end{columns}
\end{frame}

\begin{frame}{As trilhas organizam uma sequência de raciocínio}
  \begin{itemize}
    \item \textbf{Transporte de fluidos:} geometria e catálogo $\rightarrow$ regime de escoamento $\rightarrow$ perda de carga $\rightarrow$ bomba.
    \item \textbf{Reatores ideais:} componentes e estequiometria $\rightarrow$ cinética $\rightarrow$ conversão $\rightarrow$ volume ou desempenho.
    \item \textbf{Balanço de massa:} componentes $\rightarrow$ correntes $\rightarrow$ reações e reciclo $\rightarrow$ conservação e composição.
  \end{itemize}
  \vspace{0.55em}
  \begin{alertblock}{Uso didático}
    O estudante carrega um exemplo, confere hipóteses, executa o cálculo e altera uma variável por vez para observar a resposta do fenômeno.
  \end{alertblock}
\end{frame}

  \begin{columns}[T,onlytextwidth]

      \end{block}
    \column{0.32\textwidth}
      \begin{block}{Ensina}
        Cada módulo mostra entrada, equação, resultado e leitura do fenômeno.

\end{frame}
