\transitionframe{Solução proposta}{Da lacuna didática para uma plataforma modular aberta}

\begin{frame}{O que é o DCOU}
  \begin{columns}[T,onlytextwidth]
    \column{0.46\textwidth}
      \begin{block}{Produto acadêmico}
        Plataforma web gratuita para cálculo e interpretação de operações unitárias em Engenharia Química.
      \end{block}
      \begin{itemize}
        \item Interface navegável por módulos.
        \item API pública documentada pelo FastAPI.
        \item Execução local, online e
        iniciado empacotamento desktop.
      \end{itemize}
    \column{0.50\textwidth}
      \begin{block}{Modos de execução}
        \scriptsize
        \setlength{\tabcolsep}{4.5pt}
        \renewcommand{\arraystretch}{1.18}
        \begin{tabular*}{\linewidth}{@{\extracolsep{\fill}}>{\raggedright\arraybackslash}p{0.20\linewidth}>{\raggedright\arraybackslash}p{0.30\linewidth}>{\raggedright\arraybackslash}p{0.34\linewidth}}
          \textbf{Modo} & \textbf{Acesso} & \textbf{Uso} \\
          \hline
          Local & Backend local + navegador & Desenvolvimento e auditoria \\
          Online & Web publicada & Banca e demonstrações \\
          Desktop & Empacotado com backend local & Uso em máquina própria \\
        \end{tabular*}
      \end{block}
  \end{columns}
\end{frame}

\begin{frame}{Cobertura implementada}
  \small
  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
      \begin{block}{Hidráulica e bombas}
        Tubulações, schedules, acessórios, diâmetro calculado/comercial, Reynolds, atrito, perda de carga, NPSH e altura manométrica.
      \end{block}
      \begin{block}{Propriedades}
        Componentes, fluido puro, misturas, equilíbrio binário, McCabe-Thiele, envelope de fase, pressão de vapor e superfície T-P.
      \end{block}
    \column{0.48\textwidth}
      \begin{block}{Reatores}
        CSTR, PFR, reagente limitante, cinética de potência, Levenspiel, Arrhenius, perfis e reciclo.
      \end{block}
      \begin{block}{Balanço e apoio}
        Balanço com correntes, reações, splits, reciclo, gráficos, tabela, glossário e trilhas.
      \end{block}
  \end{columns}
\end{frame}

\begin{frame}{O diferencial didático é a integração}
  \begin{columns}[T,onlytextwidth]
    \column{0.30\textwidth}
      \begin{block}{Entrada}
        Formulários com unidades, catálogos e exemplos coerentes.
      \end{block}
    \column{0.30\textwidth}
      \begin{block}{Processamento}
        Backend resolve cálculos, valida hipóteses e monta modelos de gráfico.
      \end{block}
    \column{0.30\textwidth}
      \begin{block}{Interpretação}
        Tabelas, gráficos, glossário e blocos ``Como funciona'' conectam número e conceito.
      \end{block}
  \end{columns}
\end{frame}

\begin{frame}{Trilhas de aprendizagem organizam o uso}
  \begin{itemize}
    \item \textbf{Transporte de Fluidos:} Tubulações $\rightarrow$ Dimensionamento $\rightarrow$ Escoamento $\rightarrow$ Bombas.
    \item \textbf{Reatores Ideais:} Componentes $\rightarrow$ CSTR/PFR $\rightarrow$ Levenspiel e Arrhenius.
    \item \textbf{Balanço de Massa:} Componentes $\rightarrow$ Correntes $\rightarrow$ Reações $\rightarrow$ Reciclo $\rightarrow$ Resultados.
  \end{itemize}
  \vspace{0.7em}
  \begin{alertblock}{Uso em aula}
    O estudante pode carregar um exemplo, conferir hipóteses, executar o cálculo e alterar uma variável por vez.
  \end{alertblock}
\end{frame}
